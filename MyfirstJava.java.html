public class MyfirstJava {

    public static void main(String[] args) {

        // Question 1
        Shape shape = new Shape();
        shape.setColor("Blue");

        System.out.println("Shape Color: " + shape.getColor());

        // Question 2
        Rectangle rectangle = new Rectangle();
        rectangle.setColor("Red");
        rectangle.length = 10;
        rectangle.width = 5;

        System.out.println("Rectangle Information:");
        rectangle.DisplayInformation();
        System.out.println("Color: " + rectangle.getColor());

        // Question 3 - Method Overriding
        System.out.println("Method Overriding:");
        Shape shape2 = new Rectangle();
        shape2.display();

        // Question 4 - Array of Shape References
        System.out.println("Shape Array:");

        Shape[] shapes = new Shape[3];

        shapes[0] = new Shape();
        shapes[1] = new Rectangle();
        shapes[2] = new Circle();

        for (Shape s : shapes) {
            s.display();
        }

        // Question 7 - Polymorphism
        System.out.println("Polymorphism:");

        Shape s1 = new Rectangle();
        Shape s2 = new Circle();

        s1.display();
        s2.display();

        // Question 5 - super()
        System.out.println("\nSuper Example:");

        Rectangle rectangle2 = new Rectangle("Green", 20, 10);
        rectangle2.DisplayInformation();
        System.out.println("Color: " + rectangle2.getColor());
    }
}


// Parent Class
class Shape {

    String color;

    // Default constructor
    Shape() {
        color = "Unknown";
    }

    // Constructor
    Shape(String color) {
        this.color = color;
    }

    // Setter
    public void setColor(String color) {
        this.color = color;
    }

    // Getter
    public String getColor() {
        return color;
    }

    // Method that can be overridden
    public void display() {
        System.out.println("This is a shape.");
    }
}


// Child Class
class Rectangle extends Shape {

    double length;
    double width;

    // Default constructor
    Rectangle() {
        super();
        length = 0;
        width = 0;
    }

    // Constructor using super()
    Rectangle(String color, double length, double width) {
        super(color);
        this.length = length;
        this.width = width;
    }

    // Display length and width
    public void DisplayInformation() {
        System.out.println("Length: " + length);
        System.out.println("Width: " + width);
    }

    // Method overriding
    @Override
    public void display() {
        System.out.println("This is a rectangle.");
    }
}


// Another child class for polymorphism


    class Circle extends Shape {
    public void display() {
        System.out.println("This is a circle.");
    }
}
